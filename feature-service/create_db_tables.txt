-- ========== Table 1: min_price_data ==========
CREATE TABLE x_min_price_data (
    datetime TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    ticker VARCHAR(100) NOT NULL,
    price_open NUMERIC(15,6),
    price_close NUMERIC(15,6),
    price_high NUMERIC(15,6),
    price_low NUMERIC(15,6),
    price_vol NUMERIC(15,6),
    price_vol_weight_avg NUMERIC(15,6),
    CONSTRAINT x_min_price_data_pkey PRIMARY KEY (datetime, ticker)
);
